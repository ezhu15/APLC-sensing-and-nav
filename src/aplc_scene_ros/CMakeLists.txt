cmake_minimum_required(VERSION 3.8)
project(aplc_scene_ros)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
# Reuse the independently runnable converter; do not duplicate geometry logic.
ament_python_install_package(aplc_scene PACKAGE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/scene_converter/aplc_scene")
install(PROGRAMS scripts/scene_publisher scripts/quest_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch rviz DESTINATION share/${PROJECT_NAME})
install(DIRECTORY ../../maps/ DESTINATION share/${PROJECT_NAME}/maps FILES_MATCHING PATTERN "*.scene.json")
find_package(rviz_common REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(pluginlib REQUIRED)
add_library(quest_pov_view SHARED src/quest_pov_view.cpp)
ament_target_dependencies(quest_pov_view rviz_common rviz_default_plugins pluginlib)
pluginlib_export_plugin_description_file(rviz_common quest_pov_plugins.xml)
install(TARGETS quest_pov_view LIBRARY DESTINATION lib)
ament_package()
